cmake_minimum_required(VERSION 3.25)
project(MeguvizMemoryCppCheckedDevelopment LANGUAGES CXX)
find_package(Threads REQUIRED)
option(MGMEM_SANITIZE "Enable checked C++ adapter address/undefined-behavior diagnostics" OFF)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)

add_library(cpp_fixture_options INTERFACE)
target_compile_features(cpp_fixture_options INTERFACE cxx_std_17)
target_include_directories(cpp_fixture_options INTERFACE "${ROOT}/include")
target_link_libraries(cpp_fixture_options INTERFACE Threads::Threads)

if(MSVC)
  target_compile_options(cpp_fixture_options INTERFACE /W4 /permissive- /utf-8)
else()
  target_compile_options(cpp_fixture_options INTERFACE -Wall -Wextra -Wpedantic)
endif()

if(MGMEM_SANITIZE)
  if(MSVC)
    target_compile_options(cpp_fixture_options INTERFACE /fsanitize=address /Zi)
    target_link_options(cpp_fixture_options INTERFACE /INCREMENTAL:NO)
  else()
    target_compile_options(cpp_fixture_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(cpp_fixture_options INTERFACE -fsanitize=address,undefined)
  endif()
endif()

add_library(cpp_fixture_core STATIC "${ROOT}/src/mgmem.cpp")
target_link_libraries(cpp_fixture_core PUBLIC cpp_fixture_options)

add_executable(cpp_checked_import "${ROOT}/tests/cpp_checked_import_tests.cpp")
target_link_libraries(cpp_checked_import PRIVATE cpp_fixture_core)

add_executable(cpp_checked_fake "${ROOT}/tests/cpp_checked_fake_tests.cpp")
target_link_libraries(cpp_checked_fake PRIVATE cpp_fixture_options)

add_executable(cpp_legacy_consumer "${ROOT}/tests/cpp_consumer.cpp")
target_link_libraries(cpp_legacy_consumer PRIVATE cpp_fixture_core)
enable_testing()
foreach(case normal owner-failure borrow-failure destructor-failure borrow-destructor move-assignment rollback-failure concurrent arena-normal arena-failure scoped-normal scoped-failure)
  add_test(NAME cpp_checked_${case} COMMAND cpp_checked_import ${case})
  set_tests_properties(cpp_checked_${case} PROPERTIES TIMEOUT 20)
endforeach()
add_test(NAME cpp_checked_fake_unconsumed COMMAND cpp_checked_fake)
set_tests_properties(cpp_checked_fake_unconsumed PROPERTIES TIMEOUT 20)
add_test(NAME cpp_scoped_failstop_borrow COMMAND cpp_checked_fake --scoped-failstop)
set_tests_properties(cpp_scoped_failstop_borrow PROPERTIES TIMEOUT 20)
add_test(NAME cpp_legacy_consumer COMMAND cpp_legacy_consumer)
set_tests_properties(cpp_legacy_consumer PROPERTIES TIMEOUT 30)

cmake_minimum_required(VERSION 3.25)
project(MeguvizAhbUploadAllocCount LANGUAGES CXX)
find_package(MeguvizMemory CONFIG REQUIRED)
find_package(MeguvizCompute CONFIG REQUIRED)
get_target_property(compute_headers MeguvizCompute::core INTERFACE_INCLUDE_DIRECTORIES)
add_executable(ahb_upload_alloc_count ahb_upload_alloc_count.cpp ../../src/ahb_upload.cpp ../../../compute_gpu_ahb/src/gpu_ahb.cpp)
target_compile_features(ahb_upload_alloc_count PRIVATE cxx_std_17)
target_compile_definitions(ahb_upload_alloc_count PRIVATE MGMEM_GPU_AHB_TESTING)
target_include_directories(ahb_upload_alloc_count PRIVATE ../../include ../../src ../../../compute_gpu_ahb/include ${compute_headers})
target_link_libraries(ahb_upload_alloc_count PRIVATE MeguvizMemory::mgmem)
enable_testing()
add_test(NAME ahb_upload_alloc_count COMMAND ahb_upload_alloc_count)
set_tests_properties(ahb_upload_alloc_count PROPERTIES TIMEOUT 30)

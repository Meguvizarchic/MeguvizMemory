cmake_minimum_required(VERSION 3.25)
project(MeguvizAhbGuardedTests LANGUAGES CXX)
find_package(MeguvizMemory CONFIG REQUIRED)
find_package(MeguvizCompute CONFIG REQUIRED)
get_target_property(compute_headers MeguvizCompute::core INTERFACE_INCLUDE_DIRECTORIES)
add_executable(guarded_tests guarded_tests.cpp ../../src/ahb_upload.cpp
  ../../src/ahb_upload_guarded.cpp ../../../compute_gpu_ahb/src/gpu_ahb.cpp)
target_compile_features(guarded_tests PRIVATE cxx_std_17)
target_compile_definitions(guarded_tests PRIVATE MGMEM_GPU_AHB_TESTING)
target_include_directories(guarded_tests PRIVATE ../../include ../../src ../../../compute_gpu_ahb/include ${compute_headers})
target_link_libraries(guarded_tests PRIVATE MeguvizMemory::mgmem)
enable_testing()
foreach(mode IN ITEMS success failure cleanup unknown)
  add_test(NAME guarded_${mode} COMMAND guarded_tests ${mode})
  set_tests_properties(guarded_${mode} PROPERTIES TIMEOUT 30)
endforeach()

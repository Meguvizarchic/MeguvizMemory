cmake_minimum_required(VERSION 3.25)
project(MeguvizAhbUploadIntegration LANGUAGES CXX)
find_package(MeguvizMemory CONFIG REQUIRED)
find_package(MeguvizCompute CONFIG REQUIRED)
get_target_property(compute_headers MeguvizCompute::core INTERFACE_INCLUDE_DIRECTORIES)
add_executable(ahb_upload_integration integration.cpp ../../src/ahb_upload.cpp ../../../compute_gpu_ahb/src/gpu_ahb.cpp)
target_compile_features(ahb_upload_integration PRIVATE cxx_std_17)
target_compile_definitions(ahb_upload_integration PRIVATE MGMEM_GPU_AHB_TESTING)
target_include_directories(ahb_upload_integration PRIVATE ../../include ../../src ../../../compute_gpu_ahb/include ${compute_headers})
target_link_libraries(ahb_upload_integration PRIVATE MeguvizMemory::mgmem)
enable_testing()
add_test(NAME ahb_upload_actual_memory_lifetime COMMAND ahb_upload_integration)
add_test(NAME ahb_upload_actual_memory_fence_failed COMMAND ahb_upload_integration --failed)
add_test(NAME ahb_upload_actual_memory_fence_unknown COMMAND ahb_upload_integration --unknown)
add_test(NAME ahb_upload_actual_memory_fence_unverified COMMAND ahb_upload_integration --unverified)
set_tests_properties(ahb_upload_actual_memory_lifetime ahb_upload_actual_memory_fence_failed
  ahb_upload_actual_memory_fence_unknown ahb_upload_actual_memory_fence_unverified PROPERTIES TIMEOUT 30)

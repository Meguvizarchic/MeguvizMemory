cmake_minimum_required(VERSION 3.25)
project(InstalledMediaBridgeConsumer LANGUAGES C CXX)
if(REQUIRE_ARENA)
  find_package(MeguvizMemoryMediaReadBridge CONFIG REQUIRED COMPONENTS arena_read_bridge)
else()
  find_package(MeguvizMemoryMediaReadBridge CONFIG REQUIRED COMPONENTS read_bridge)
endif()
add_executable(media_bridge_system_consumer system_consumer.c)
target_link_libraries(media_bridge_system_consumer PRIVATE MeguvizMemoryMediaReadBridge::read_bridge)
add_executable(media_short_system_c short_system.c)
target_compile_features(media_short_system_c PRIVATE c_std_11)
target_link_libraries(media_short_system_c PRIVATE MeguvizMemoryMediaReadBridge::read_bridge)
add_executable(media_short_system_cpp short_system_cpp.cpp)
target_compile_features(media_short_system_cpp PRIVATE cxx_std_17)
target_link_libraries(media_short_system_cpp PRIVATE MeguvizMemoryMediaReadBridge::read_bridge)

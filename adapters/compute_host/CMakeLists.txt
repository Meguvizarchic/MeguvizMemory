cmake_minimum_required(VERSION 3.25)
project(MeguvizMemoryComputeHostBridge VERSION 1.2.2 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(MeguvizMemory CONFIG REQUIRED)
find_package(MeguvizCompute CONFIG REQUIRED)
include(CheckCXXSourceCompiles)
set(_host_bridge_saved_libraries "${CMAKE_REQUIRED_LIBRARIES}")
set(CMAKE_REQUIRED_LIBRARIES MeguvizCompute::core)
unset(MGC_HAS_HOST_IMPORT CACHE)
check_cxx_source_compiles("#include <meguviz/compute.h>
int main() { return (int)mgc_resource_import_host_v1(nullptr, nullptr, nullptr, nullptr); }" MGC_HAS_HOST_IMPORT)
set(CMAKE_REQUIRED_LIBRARIES "${_host_bridge_saved_libraries}")
unset(_host_bridge_saved_libraries)
if(NOT MGC_HAS_HOST_IMPORT)
  message(FATAL_ERROR "Compute SDK requires mgc_resource_import_host_v1")
endif()
add_library(mgmem_compute_host_bridge STATIC src/compute_host_bridge.cpp src/compute_transfer.cpp src/compute_stage.cpp)
target_compile_features(mgmem_compute_host_bridge PUBLIC cxx_std_17)
target_include_directories(mgmem_compute_host_bridge PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(mgmem_compute_host_bridge PUBLIC MeguvizMemory::mgmem MeguvizCompute::core)
set_target_properties(mgmem_compute_host_bridge PROPERTIES EXPORT_NAME host_bridge CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
add_library(MeguvizMemoryComputeHostBridge::host_bridge ALIAS mgmem_compute_host_bridge)
install(TARGETS mgmem_compute_host_bridge EXPORT MeguvizMemoryComputeHostBridgeTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT MeguvizMemoryComputeHostBridgeTargets NAMESPACE MeguvizMemoryComputeHostBridge:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MeguvizMemoryComputeHostBridge)
configure_package_config_file(cmake/MeguvizMemoryComputeHostBridgeConfig.cmake.in
  ${CMAKE_CURRENT_BINARY_DIR}/MeguvizMemoryComputeHostBridgeConfig.cmake
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MeguvizMemoryComputeHostBridge)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/MeguvizMemoryComputeHostBridgeConfigVersion.cmake
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/MeguvizMemoryComputeHostBridgeConfig.cmake
  ${CMAKE_CURRENT_BINARY_DIR}/MeguvizMemoryComputeHostBridgeConfigVersion.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MeguvizMemoryComputeHostBridge)
option(MGMEM_COMPUTE_HOST_BRIDGE_BUILD_TESTS "Build scoped host-resource bridge tests" ON)

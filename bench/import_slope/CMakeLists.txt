cmake_minimum_required(VERSION 3.25)
project(MeguvizMemoryImportSlope LANGUAGES CXX)
find_package(MeguvizMemory CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(mgmem_import_slope import_slope.cpp)
target_compile_features(mgmem_import_slope PRIVATE cxx_std_17)
target_link_libraries(mgmem_import_slope PRIVATE MeguvizMemory::mgmem Threads::Threads ${CMAKE_DL_LIBS})
if(MSVC)
  target_compile_options(mgmem_import_slope PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(mgmem_import_slope PRIVATE -Wall -Wextra -Wpedantic)
endif()

cmake_minimum_required(VERSION 3.25)
project(MeguvizMemoryClaimRelease LANGUAGES CXX)
find_package(MeguvizMemory CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(mgmem_claim_release ../claim_release.cpp)
target_compile_features(mgmem_claim_release PRIVATE cxx_std_17)
target_link_libraries(mgmem_claim_release PRIVATE MeguvizMemory::mgmem Threads::Threads ${CMAKE_DL_LIBS})
if(MSVC)
  target_compile_options(mgmem_claim_release PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(mgmem_claim_release PRIVATE -Wall -Wextra -Wpedantic)
endif()
